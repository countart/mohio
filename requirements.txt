# ── Parser ───────────────────────────────────────────────────
lark>=1.1.0

# ── AI Providers ─────────────────────────────────────────────
# Anthropic (Claude) — required. Default provider for ai.decide and
# ai.create text. Bring your own key: set ANTHROPIC_API_KEY=sk-ant-...
anthropic>=0.96.0

# GPT's client-side tokenizer, for ai.agent's pre-call cost-cap estimate
# (2026-08-06). OpenAI has no countTokens-style API the way Claude
# (messages.count_tokens) and Gemini (:countTokens) do, so this is the
# only accurate way to estimate a GPT call's input cost before it fires.
# A character-count heuristic was considered and rejected: pre-call
# refusal exists specifically to catch a breach before it happens, and an
# inaccurate estimate defeats that either direction (wrongly refuses a
# valid call, or wrongly lets a real breach through).
tiktoken>=0.13.0

# OpenAI and Gemini providers (used in ai.connect chains and ai.create
# image/video) are called over plain HTTP — NO extra SDK install required
# for the calls themselves (tiktoken above is a tokenizer, not an SDK).
# Just set the env keys when you actually use those providers:
#   OPENAI_API_KEY   — gpt-*, o-series, gpt-image/dall-e, sora
#   GEMINI_API_KEY   (or GOOGLE_API_KEY) — gemini-*, imagen, veo
# A missing key only errors if that provider is called directly; in a
# fallback chain a keyless provider is simply skipped.

# ── Auth / Security ──────────────────────────────────────────
# Password hashing for `hash bcrypt` / `verify`. Imported lazily inside
# a try/except, so the deploy still boots if it is absent — but bcrypt
# hashing will not work until this is installed.
bcrypt>=4.0

# Field / at-rest encryption (sealed [phi]/[pii] fields, key-provider seam). Imported lazily,
# so the deploy still boots without it — but encrypted writes fail with "No module named
# 'cryptography'" and PHI/PII would not be sealed. Required for any encryption-backed compliance.
cryptography>=49.0.0

# ── Upload safety ────────────────────────────────────────────
# HTML sanitizer for `accept html` uploads. An uploaded page is cleaned
# before it is stored, so hidden script cannot run against whoever opens
# it later. Without this, an HTML upload is REFUSED rather than stored
# unsanitized -- so a deploy missing it fails loud instead of silently
# serving whatever was uploaded.
# NOTE: bleach was the usual choice, but it was retired on 2026-06-05
# and gets no further security fixes. nh3 (Ammonia bindings) is maintained.
